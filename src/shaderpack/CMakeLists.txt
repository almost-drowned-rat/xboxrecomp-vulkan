# The user's slang shader packs for the shader module (shader_pack.c): zips in
# media/shaders/ extracted once to data/shaders/, the preset index. Host code,
# no guest knowledge, no Vulkan; file I/O and the worker thread through SDL3.
# Inflate: miniz's tinfl, vendored in third_party/miniz (miniz 3.1.2, MIT; see
# NOTICE) - only miniz_tinfl.c is compiled, the other miniz headers are there
# because miniz_tinfl.c includes miniz.h (its deflate / zip / zlib parts are
# switched off below).

add_library(xbox_shaderpack STATIC
    zip_read.c
    shader_pack.c
    third_party/miniz/miniz_tinfl.c
)
target_include_directories(xbox_shaderpack PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_include_directories(xbox_shaderpack PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/third_party/miniz)
# miniz_export.h is generated by miniz's own build; here it is empty (static)
file(WRITE ${CMAKE_CURRENT_BINARY_DIR}/miniz_export.h "#define MINIZ_EXPORT\n")
target_include_directories(xbox_shaderpack PRIVATE ${CMAKE_CURRENT_BINARY_DIR})
target_compile_definitions(xbox_shaderpack PRIVATE MINIZ_NO_STDIO MINIZ_NO_TIME MINIZ_NO_DEFLATE_APIS
    MINIZ_NO_ARCHIVE_APIS MINIZ_NO_ZLIB_APIS MINIZ_NO_ZLIB_COMPATIBLE_NAMES)

find_package(SDL3 CONFIG REQUIRED)
target_link_libraries(xbox_shaderpack PUBLIC SDL3::SDL3)

if(MSVC)
    target_compile_options(xbox_shaderpack PRIVATE /W4 /O2)
else()
    target_compile_options(xbox_shaderpack PRIVATE -Wall -Wextra -O2)
    set_source_files_properties(third_party/miniz/miniz_tinfl.c PROPERTIES COMPILE_OPTIONS "-w")
endif()
