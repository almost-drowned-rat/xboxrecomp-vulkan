cmake_minimum_required(VERSION 3.20)
project(video_aspect_test C)
set(CMAKE_C_STANDARD 11)
add_subdirectory(../.. runtime)
add_executable(video_aspect_test test_main.c)
target_include_directories(video_aspect_test PRIVATE ../../src/kernel)
target_link_libraries(video_aspect_test PRIVATE xboxrecomp)
if(NOT MSVC)
    target_link_libraries(video_aspect_test PRIVATE m)
endif()
enable_testing()
add_test(NAME video_aspect_fit COMMAND video_aspect_test fit)
add_test(NAME video_aspect_keep COMMAND video_aspect_test keep)
add_test(NAME video_aspect_env COMMAND video_aspect_test env)
add_test(NAME video_aspect_other COMMAND video_aspect_test other)
